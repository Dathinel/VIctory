+#[test]
+fn bcpack_interleave_on_the_gpu() {
+    // The first symbol occupies even bits; the second occupies odd bits.
+    let mut body = vec![vop1(0x001, 0, 128)];
+    for bit in 0u16..4 {
+        for (source, position) in [(1, 2 * bit), (2, 2 * bit + 1)] {
+            body.extend(vop3(0x148, 3, vgpr(source), 128 + bit, 129)); // bfe_u32
+            body.extend(vop3(0x36f, 0, vgpr(3), 128 + position, vgpr(0))); // lshl_or_b32
+        }
+    }
+    let shader = build_valu_cs(&body);
+    if let Some(dir) = std::env::var_os("BCPACK_SHADER_OUTPUT") {
+        std::fs::write(std::path::Path::new(&dir).join("interleave.dxil"), &shader.dxil).unwrap();
+    }
+    // Invert the library mapping to generate inputs from every possible result.
+    let mut cases = Vec::new();
+    for byte in 0u32..256 {
+        let compact = |word: u32| (0..4).fold(0, |acc, bit| acc | ((word >> (2 * bit)) & 1) << bit);
+        cases.push([compact(byte), compact(byte >> 1), 0, 0]);
+    }
+    // Nonzero high nibbles check the masking implicit in the original u8 function.
+    cases.extend([[0xff, 0xf0, 0, 0], [0xa0, 0x5f, 0, 0]]);
+    let mut actual = std::collections::BTreeMap::new();
+    run_cases(&shader, &cases, |backend, i, _, got| {
+        let expected = match i { 256 => 0x55, 257 => 0xaa, _ => i as u32 };
+        assert_eq!(got, expected, "{backend}: case {i}");
+        *actual.entry(backend.to_string()).or_insert(0usize) += 1;
+    });
+    eprintln!("checked: {actual:?}");
+    assert!(!actual.is_empty());
+    assert!(actual.values().all(|&n| n == cases.len()));
+}
