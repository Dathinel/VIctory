 /// VOP1 `op vdst, src0` (`src0`: 0..105 = SGPR, 256..511 = VGPR).
 const fn vop1(op: u16, vdst: u8, src0: u16) -> u32 {
     0x7E00_0000 | (vdst as u32) << 17 | (op as u32) << 9 | (src0 as u32 & 0x1FF)
+}
+
+#[test]
+#[ignore = "requires synthetic BC4 fixtures, DXC output and D3D12"]
+fn bc4_sized_gpu() {
+    use super::host::Stride;
+    use std::{fs, path::PathBuf};
+
+    let dir = PathBuf::from(std::env::var_os("BC1_TEST_DIR").expect("fixture directory"));
+    let mut shader = build_valu_cs(&[vop1(1, 0, 128)]);
+    shader.dxil = fs::read(dir.join("bc1-records.dxil")).unwrap();
+    for (device, warp) in [("hardware", false), ("warp", true)] {
+        let host = Host::new(warp).unwrap();
+        let (root, pipeline) = host.compute_pipeline(&shader).unwrap();
+        for slot in [2, 260] {
+            let input = fs::read(dir.join(format!("input-{slot}.bin"))).unwrap();
+            let expected = fs::read(dir.join(format!("expected-{slot}.bin"))).unwrap();
+            assert!(input.len().is_multiple_of(4) && expected.len().is_multiple_of(4));
+            let data: Vec<u32> = input
+                .chunks_exact(4)
+                .map(|b| u32::from_le_bytes(b.try_into().unwrap()))
+                .collect();
+            let words = u32::try_from(expected.len() / 4).unwrap();
+            let views = [
+                View::Buffer {
+                    data: &data,
+                    stride: Stride::Raw,
+                },
+                View::RawUav(words.checked_add(2).unwrap()),
+            ];
+            let output = host
+                .dispatch(
+                    &root,
+                    &pipeline,
+                    &Dispatch {
+                        roots: vec![(
+                            0,
+                            Root::Constants(vec![0; shader.user_data_dwords().max(1) as usize]),
+                        )],
+                        views: &views,
+                        samplers: &[],
+                        groups: [1, 1, 1],
+                    },
+                )
+                .unwrap();
+            let actual: Vec<u8> = output
+                .last()
+                .unwrap()
+                .iter()
+                .flat_map(|v| v.to_le_bytes())
+                .collect();
+            assert_eq!(actual.len(), expected.len() + 8);
+            if let Some(offset) = actual.iter().zip(&expected).position(|(a, b)| a != b) {
+                panic!("{device}: mismatch in slot {slot} at byte {offset}");
+            }
+            assert_eq!(&actual[expected.len()..], &[0; 8], "{device}: guard");
+            eprintln!(
+                "{device}: slot {slot} matched {} bytes and guard",
+                expected.len()
+            );
+        }
+    }
 }
